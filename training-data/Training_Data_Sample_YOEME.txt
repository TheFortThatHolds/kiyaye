import numpy as np

class AttentionMechanism:
   """
   Scaled Dot-Product Attention mekanismo implementaroa.
   Ref: "Attention Is All You Need" (Vaswani et al., 2017)
   """
   def __init__(self, d_k):
       # d_k llave vectoram dimensión machiria.
       # Ini dot producto eskalarpo usariwa gradientes desaparecienteam prevenirpo.
       self.d_k = d_k

   def forward(self, query, key, value, mask=None):
       """
       Atención puntuación aman valores suma ponderada reknao.
       
       Argumentos:
           query (np.array): Forma (batch, seq_len, d_k)
           key (np.array): Forma (batch, seq_len, d_k)
           value (np.array): Forma (batch, seq_len, d_v)
           mask (np.array, opcional): Forma (batch, seq_len, seq_len). 
                                      1 mask machiria, 0 koptia machiria.
       
       Yeu Weyemta:
           toradh, atención_pesos
       """
       # 1. MatMul: Puntuación krudo reknao (Query x Key^T)
       # Key uttia gonak dimensión transponer multiplikasionpo
       scores = np.matmul(query, key.swapaxes(-2, -1))
       
       # 2. Eskala: sqrt(d_k) betchi'ibo partia
       scores = scores / np.sqrt(self.d_k)
       
       # 3. Mask: Mask aplicar au respo (mask posiciones -infinito settia)
       if mask is not None:
           # Itepo número negativo bwe'u usariwa -infinito Softmaxpo simularpo
           scores = np.where(mask == 1, -1e9, scores)
       
       # 4. Softmax: Puntuaciones probabilidadespo normaliza
       attention_weights = self._softmax(scores)
       
       # 5. MatMul: Valores suma ponderada
       output = np.matmul(attention_weights, value)
       
       return output, attention_weights

   def _softmax(self, x):
       """Numerikamente estable softmax implementasio."""
       e_x = np.exp(x - np.max(x, axis=-1, keepdims=True))
       return e_x / np.sum(e_x, axis=-1, keepdims=True)

# Driver Código uso demostrarpo
if __name__ == "__main__":
   # Dimensiones simularpo: Batch=1, SeqLen=3, Dimensión=4

Fierro-Karbono Ekilibrio & Asero Taewai Tratamiento

Asero primariamente fierro (Fe) aman karbono (C) aleación, típikamente 0.002% aman 2.1% karbono peso betchi'ibo kontiene. Asero propiedades mekanikam mikroestructura betchi'ibo determina, ini taewai tratamiento prokesoam betchi'ibo manipulariatua kepar annealing, normalizing, quenching, aman tempering.

1. Asero Fasesam:
   * Ferrita (\alpha-fierro): Body-Centered Cubic (BCC) kristal estructura. Suave, dúktil, aman magnétiko temperatura ambiente. Karbono chikti disuelvewa (max 0.022% 727°Cpo).
   * Austenita (\gamma-fierro): Face-Centered Cubic (FCC) estructura. Temperatura alto forma (727°C wechiapo). Ka magnétiko aman karbono bwe'u disuelvewa (2.14% asta 1147°Cpo).
   * Sementita (Fe_3C): Kompuesto intermetáliko. Bwe'uka duro aman bwe'uka frágil.
   * Perlita: Laminar (capas) estructura ferrita aman sementita capas alternadas. Austenita sualem enfriaktuka forma.
2. Quenching Mekanismo: Austenita bwe'uka venam temperatura ambiente enfriaktuka (quenched), karbono átomos tiempo ka aune latiseko sementita formarpo difundir. Au lugar, atrapados yeu weyek, kristal latise cizallaroa. Ini FCC austenita Martensitako transforma, Body-Centered Tetragonal (BCT) estructura.
   * Propiedad: Martensita asero forma más dura aman más fuerte pero bwe'uka frágil.
   * Sistema Cheka: Puro martensita kuchillo impaktopo tet'etiwane.
3. Tempering: Martensita fragilidad korregirpo, asero taewai we'epusia temperatura eutektoido punto yetukupo (usualmente 150°C aman 600°C wepapo). Ini karbono difusión chikti permitewa, tensiones internas aliviaroa aman martensita tempered martensitako (ferrita + sementita partíkulas finas) transformaroa. Ini dureza benaka tenacidad restauraroa.
