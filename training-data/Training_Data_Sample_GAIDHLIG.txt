import numpy as np

class AttentionMechanism:
   """
   A' cur an gnìomh inneal Scaled Dot-Product Attention.
   Iomradh: "Attention Is All You Need" (Vaswani et al., 2017)
   """
   def __init__(self, d_k):
       # Tha d_k a' riochdachadh meud nan vectoran iuchrach.
       # Thèid a chleachdadh airson na toraidhean dot a sgèileadh gus casg a chur air gradients a' crìonadh.
       self.d_k = d_k

   def forward(self, query, key, value, mask=None):
       """
       Àireamhaich sgòran aire agus suim chuideamach de luachan.
       
       Argamaidean:
           query (np.array): Cumadh (batch, seq_len, d_k)
           key (np.array): Cumadh (batch, seq_len, d_k)
           value (np.array): Cumadh (batch, seq_len, d_v)
           mask (np.array, roghainneil): Cumadh (batch, seq_len, seq_len). 
                                         Tha 1 a' ciallachadh masg, tha 0 a' ciallachadh cùm.
       
       A' tilleadh:
           toradh, cuideaman_aire
       """
       # 1. MatMul: Àireamhaich sgòran amh (Query x Key^T)
       # Cuir mun cuairt an dà thomhas mu dheireadh de Key airson iomadachadh
       scores = np.matmul(query, key.swapaxes(-2, -1))
       
       # 2. Sgèile: Roinn le sqrt(d_k)
       scores = scores / np.sqrt(self.d_k)
       
       # 3. Masg: Cuir masg an sàs ma thèid a thoirt seachad (suidhich ionadan masgaichte gu -neo-chrìochnachd)
       if mask is not None:
           # Tha sinn a' cleachdadh àireamh àicheil mòr gus -neo-chrìochnachd a shamhlachadh ann an Softmax
           scores = np.where(mask == 1, -1e9, scores)
       
       # 4. Softmax: Àbhaisteach sgòran gu coltasan
       attention_weights = self._softmax(scores)
       
       # 5. MatMul: Suim chuideamach de Luachan
       output = np.matmul(attention_weights, value)
       
       return output, attention_weights

   def _softmax(self, x):
       """Cur an gnìomh softmax seasmhach gu àireamhach."""
       e_x = np.exp(x - np.max(x, axis=-1, keepdims=True))
       return e_x / np.sum(e_x, axis=-1, keepdims=True)

# Còd Draibhear gus cleachdadh a nochdadh

Tha stàilinn gu prìomhach na coimeasgachadh de dh'iarann (Fe) agus gual (C), mar as trice le eadar 0.002% agus 2.1% gual a rèir cuideam. Tha feartan meacanaigeach stàilinn air an dearbhadh le a meanbh-structar, a ghabhas atharrachadh tro phròiseasan làimhseachadh teasa leithid annealing, normalizing, quenching, agus tempering.

1. Na h-Ìrean de Stàilinn:
   * Ferite (α-iarann): Structar criostal BCC. Tha e bog, sùbailte, agus magnetach aig teòthachd seòmair. Tha e a' sgaoileadh glè bheag guail (0.022% aig a' char as àirde aig 727°C).
   * Austenite (γ-iarann): Structar FCC. Bidh e a' cruthachadh aig teòthachdan àrda (os cionn 727°C). Chan eil e magnetach agus 's urrainn dha barrachd guail a sgaoileadh (suas ri 2.14% aig 1147°C).
   * Ceimentite (Fe₃C): Todhar eadar-mheatailt. Tha e fìor chruaidh agus brisg.
   * Pearlite: Structar lamalaireach de shreathan ferite agus ceimentite. Bidh e a' cruthachadh nuair a dh'fhuaraicheas austenite gu slaodach.
2. Am Meacanism Quenching: Ma thèid austenite fhuarachadh gu math luath (quenched) gu teòthachd seòmair, chan eil ùine aig na dadam guail sgaoileadh a-mach às an laitis gus ceimentite a chruthachadh. An àite sin, bidh iad air an glacadh, a' geàrradh an laitis criostal. Tha seo ag atharrachadh an FCC austenite gu Martensite, structar BCT.
   * Feart: Is e martensite am foirm as cruaidhe agus as làidire de stàilinn ach tha e anabarrach brisg.
   * Sgrùdadh Siostam: Bhristeadh lann phur mhartensitic nuair a bhuaileadh e.
3. Tempering: Gus brìosgachd martensite a cheartachadh, thèid an stàilinn ath-theasachadh gu teòthachd fo phuing eutectoid (mar as trice eadar 150°C agus 600°C). Tha seo a' ceadachadh beagan sgaoileadh guail, a' faothachadh stressanan a-staigh agus ag atharrachadh cuid de mhartensite gu martensite tempered. Tha seo ag ath-stèidheachadh lùths air cosgais cuid de chruadhas.
